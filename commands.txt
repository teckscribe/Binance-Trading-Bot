# CSB Bot Management Commands (Ubuntu 24.04)

# Status
sudo systemctl status csb csb-bot csb-web csb-discord csb-ngrok

# Restart services
sudo systemctl restart csb
sudo systemctl restart csb-bot
sudo systemctl restart csb-web
sudo systemctl restart csb-discord
sudo systemctl restart csb-ngrok

# Logs
sudo journalctl -u csb -f
sudo journalctl -u csb-bot -f
sudo journalctl -u csb-web -f
sudo journalctl -u csb-discord -f

# Git Pull Updates
git pull origin main
sudo systemctl restart csb csb-bot csb-web csb-discord csb-ngrok
